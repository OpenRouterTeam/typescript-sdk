### `models.STTInlineInputAudio`

```typescript
const value: models.STTInlineInputAudio = {
  data: "UklGRiQA...",
  format: "wav",
};
```

### `models.STTUrlInputAudio`

```typescript
const value: models.STTUrlInputAudio = {
  url: "https://example.com/meeting.mp3",
};
```

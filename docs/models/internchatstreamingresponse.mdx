title: "InternChatStreamingResponse"

A server-sent event carrying one chunk. The stream ends with `data: [DONE]`.

## Example Usage

```typescript
import { InternChatStreamingResponse } from "@openrouter/sdk/models";

let value: InternChatStreamingResponse = {
  data: {
    choices: [
      {
        delta: {},
        finishReason: null,
        index: 0,
      },
    ],
    created: 1789537541,
    id: "chatcmpl-f727571a-3bad-4e0d-8a9e-f18f8cda9750",
    model: "openrouter/intern",
    object: "chat.completion.chunk",
  },
};
```

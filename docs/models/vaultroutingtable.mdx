title: "VaultRoutingTable"
---

The intern's routing table, for the intern's own key only; null for any other caller or when the vault could not build it in time. The keys still list.

## Example Usage

```typescript
import { VaultRoutingTable } from "@openrouter/sdk/models";

let value: VaultRoutingTable = {
  agentId: "<id>",
  customOrigins: [],
  github: {
    apps: [
      {
        appId: "<id>",
        appSlug: "<value>",
        installations: [],
        memberLinks: [],
        scope: "workspace",
      },
    ],
    tokens: [
      {
        holdsToken: false,
        login: "Concepcion.Brekke95",
        owner: "<value>",
        repositoryCount: 274676,
        scope: "intern",
      },
    ],
  },
  google: {
    accounts: [],
  },
  hostRules: [
    {
      holdsSecret: false,
      host: "splendid-forager.org",
      secretName: "<value>",
    },
  ],
  mcpOrigins: [
    {
      accountKey: "<value>",
      host: "dark-viability.info",
      isPersonal: true,
      routes: [],
      status: {
        failure: "<value>",
        state: "failing",
      },
    },
  ],
  readAt: new Date("2024-12-01T18:11:20.418Z"),
  secrets: [],
  slack: {
    memberLinks: [
      {
        holdsToken: false,
        slackUserId: "<id>",
      },
    ],
  },
};
```

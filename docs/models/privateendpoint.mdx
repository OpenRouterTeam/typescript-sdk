  id: "5b1c4c4e-7d0a-4a8e-9f3a-2d6c1b0e8a11",

    completion: "0.00001",
    prompt: "0.0000025",
  },
  providerName: "Azure",
  providerSlug: "azure",
  settings: {
    completions: {
      chat: true,
      text: false,
    },
    contextLength: 128000,
    excludedParameters: [
      "logit_bias",
    ],
    inputs: {
      audio: false,
      base64Files: null,
      base64Images: true,
      fileUrls: null,
      imageUrls: true,
    },
    maxCompletionTokens: 16384,
    maxPromptTokens: null,

      returnMechanism: "reasoning",

    "tools",
    "tool_choice",
    "response_format",
  ],
  upstreamModelId: "gpt-4o-prod",
};

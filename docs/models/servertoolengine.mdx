## Example Usage

```typescript
import { ServerToolEngine } from "@openrouter/sdk/models";

let value: ServerToolEngine = {
  byok: "none",
  dataRegions: [
    "global",
    "us",
  ],
  defaultMode: "auto",
  executedBy: "openrouter",
  id: "exa",
  name: "Exa",
  pricing: [
    {
      maxUnitsPerCall: 769230,
      mode: "auto",
      price: "0.007",
      unit: "request",
    },
  ],
  pricingDocUrl: null,
  pricingSource: "openrouter",
};
```

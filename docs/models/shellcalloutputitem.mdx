Output from a shell command execution (newer variant)

## Example Usage

```typescript
import { ShellCallOutputItem } from "@openrouter/sdk/models";

let value: ShellCallOutputItem = {
  callId: "call-abc123",
  output: [
    {
      outcome: {
        exitCode: 0,
        type: "exit",
      },
      stderr: "",
      stdout: "total 0\n",
    },
  ],
  type: "shell_call_output",
};
```

An endpoint whose data policy does not retain prompts by default

```typescript
import { PublicZdrEndpoint } from "@openrouter/sdk/models";

let value: PublicZdrEndpoint = {
  contextLength: 1048576,
  latencyLast30m: null,
  maxCompletionTokens: 65535,
  maxPromptTokens: null,
  modelId: "google/gemini-2.5-flash",
  modelName: "Gemini 2.5 Flash",
  name: "Google: Gemini 2.5 Flash",
  nativeTools: {},
  pricing: {
    completion: "0.0000025",
    prompt: "0.0000003",
  },
  providerName: "Google",
  quantization: null,
  serviceTier: null,
  supportedParameters: [
    "temperature",
    "top_p",
    "max_tokens",
  ],
  supportsDiarization: false,
  supportsImplicitCaching: true,
  supportsToolChoice: {
    auto: true,
    function: true,
    none: true,
    required: true,
  },
  tag: "google-vertex",
  throughputLast30m: null,
  uptimeLast1d: 99.8,
  uptimeLast30m: 99.5,
  uptimeLast5m: 100,
  safetyRetention: true,
};
```

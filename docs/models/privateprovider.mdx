let value: PrivateProvider = {
  baseUrl: "https://inference.example.com/v1",
  createdAt: "2026-10-06T12:00:00.000Z",

    promptRetentionDays: null,
    retainsPrompts: false,
    training: false,
  },
  displayName: "Acme",
  name: "Acme",
  slug: "acme",
};

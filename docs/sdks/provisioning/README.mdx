## provisionOrganization

Provision an organization for a Connect client

### Example Usage

```typescript
import { OpenRouter } from "@openrouter/sdk";

const openRouter = new OpenRouter({
  httpReferer: "<value>",
  appTitle: "<value>",
  appCategories: "<value>",
  apiKey: process.env["OPENROUTER_API_KEY"] ?? "",
});

async function run() {
  const result = await openRouter.provisioning.provisionOrganization({
    requestBody: {
      email: "Arlene23@gmail.com",
      externalId: "<id>",
      name: "<value>",
      paymentToken: "<value>",
      scopes: [
        "observability_destinations_write",
      ],
    },
  });

  console.log(result);
}

run();
```
